\newcommand{\tableColored}[4]{% n cols, cw, lw, rh
\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm]
  \def\n{#1}
  \def\cw{#2}
  \def\lw{#3}
  \def\rh{#4}
  \pgfmathsetmacro{\W}{\lw+\n*\cw}
  \pgfmathsetmacro{\H}{2*\rh}

  \fill[montorange,rounded corners=5pt] (0,\rh) rectangle (\W,\H);
  \fill[montblue,rounded corners=5pt] (0,0) rectangle (\W,\rh);

  \draw[softgray,line width=0.9pt,rounded corners=4pt] (0,0) rectangle (\W,\H);
  \draw[softgray,line width=0.9pt] (0,\rh)--(\W,\rh);
  \draw[softgray,line width=0.9pt] (\lw,0)--(\lw,\H);
  \foreach \i in {1,...,#1}{
    \pgfmathsetmacro{\x}{\lw+\i*\cw}
    \draw[softgray,line width=0.55pt] (\x,0)--(\x,\H);
  }

  \leftCellOne{0.18}{1.50*\rh}{toporange}
  \leftCellTwo{0.18}{0.50*\rh}{bottomblue}

  % coefficient schema, much lower, separated and readable
  \node[anchor=west,font=\large\bfseries\color{softgray}] at (0,-3.45) {Coefficient de proportionnalité :};
  \node[anchor=west,font=\large\bfseries,text=toporange] at (7.15,-3.45) {Grandeur 1};
  \node[anchor=west,font=\large] at (10.15,-3.45) {$\times$};
  \draw[softgray,densely dotted,line width=0.55pt] (10.95,-3.45)--(14.15,-3.45);
  \node[anchor=west,font=\large] at (14.55,-3.45) {$=$};
  \node[anchor=west,font=\large\bfseries,text=bottomblue] at (15.10,-3.45) {Grandeur 2};

  \node[anchor=west,font=\large\bfseries,text=bottomblue] at (7.15,-5.05) {Grandeur 2};
  \node[anchor=west,font=\large] at (10.15,-5.05) {$\times$};
  \draw[softgray,densely dotted,line width=0.55pt] (10.95,-5.05)--(14.15,-5.05);
  \node[anchor=west,font=\large] at (14.55,-5.05) {$=$};
  \node[anchor=west,font=\large\bfseries,text=toporange] at (15.10,-5.05) {Grandeur 1};
\end{tikzpicture}
\end{center}
}
