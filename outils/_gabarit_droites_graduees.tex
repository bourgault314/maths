\newcommand{\DroiteReguliere}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \useasboundingbox (0,-0.95) rectangle (28.35,0.95);
  \draw[lineInk,line width=1.25pt,-{Latex[length=3.2mm,width=2.35mm]}]
    (0.15,0) -- (28.10,0);
  \foreach \i in {0,...,26}{
    \pgfmathsetmacro{\x}{1+\i}
    \draw[lineInk,line width=0.95pt] (\x,-0.34) -- (\x,0.34);
  }
\end{tikzpicture}%
}

\newcommand{\DroiteQuarts}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \useasboundingbox (0,-0.95) rectangle (28.35,0.95);
  \draw[lineInk,line width=1.25pt,-{Latex[length=3.2mm,width=2.35mm]}]
    (0.15,0) -- (28.10,0);
  \foreach \i in {0,...,26}{
    \pgfmathsetmacro{\x}{1+\i}
    \pgfmathtruncatemacro{\reste}{mod(\i,4)}
    \ifnum\reste=0
      \draw[lineInk,line width=1.05pt] (\x,-0.47) -- (\x,0.47);
    \else
      \pgfmathtruncatemacro{\demi}{mod(\i,2)}
      \ifnum\demi=0
        \draw[lineInk,line width=0.95pt] (\x,-0.35) -- (\x,0.35);
      \else
        \draw[lineInk,line width=0.85pt] (\x,-0.25) -- (\x,0.25);
      \fi
    \fi
  }
\end{tikzpicture}%
}

\newcommand{\DroiteDixiemes}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \useasboundingbox (0,-0.95) rectangle (28.35,0.95);
  \draw[lineInk,line width=1.25pt,-{Latex[length=3.2mm,width=2.35mm]}]
    (0.15,0) -- (28.10,0);
  \foreach \i in {0,...,26}{
    \pgfmathsetmacro{\x}{1+\i}
    \pgfmathtruncatemacro{\resteDix}{mod(\i,10)}
    \pgfmathtruncatemacro{\resteCinq}{mod(\i,5)}
    \ifnum\resteDix=0
      \draw[lineInk,line width=1.05pt] (\x,-0.47) -- (\x,0.47);
    \else
      \ifnum\resteCinq=0
        \draw[lineInk,line width=0.95pt] (\x,-0.35) -- (\x,0.35);
      \else
        \draw[lineInk,line width=0.80pt] (\x,-0.22) -- (\x,0.22);
      \fi
    \fi
  }
\end{tikzpicture}%
}
