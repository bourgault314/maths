\newcommand{\DroiteCentimetres}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \useasboundingbox (0,-0.95) rectangle (28.35,0.95);
  \draw[lineInk,line width=1.25pt,-{Latex[length=3.2mm,width=2.35mm]}]
    (0.15,0) -- (28.10,0);

  }
\end{tikzpicture}%
}

\newcommand{\DroiteDemiCentimetres}{%
\begin{tikzpicture}[x=1cm,y=1cm]

  }
\end{tikzpicture}%
}

\newcommand{\DroiteMillimetres}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \useasboundingbox (0,-0.95) rectangle (28.35,0.95);
  \draw[lineInk,line width=1.25pt,-{Latex[length=3.2mm,width=2.35mm]}]
    (0.15,0) -- (28.10,0);
  \foreach \i in {0,...,270}{
    \pgfmathsetmacro{\x}{0.5+0.1*\i}
    \pgfmathtruncatemacro{\resteDix}{mod(\i,10)}
    \pgfmathtruncatemacro{\resteCinq}{mod(\i,5)}
    \ifnum\resteDix=0
      \draw[lineInk,line width=0.90pt] (\x,-0.45) -- (\x,0.45);
    \else
      \ifnum\resteCinq=0
        \draw[lineInk,line width=0.80pt] (\x,-0.31) -- (\x,0.31);
      \else
        \draw[lineInk,line width=0.55pt] (\x,-0.18) -- (\x,0.18);
      \fi
    \fi
  }
\end{tikzpicture}%
}
