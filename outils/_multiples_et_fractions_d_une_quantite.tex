% ==========================================
% MACROS : GABARITS REMPLIS (COLONNE GAUCHE)
% ==========================================
\newcommand{\gabaritRempli}[4]{%
    \begin{tikzpicture}[scale=1]
        \useasboundingbox (0, -0.3) rectangle (\Largeur, 2.7);
        \pgfmathsetmacro{\w}{\Largeur/#1}

        % Bloc total (Haut)
        \draw[thick, fill=#2!20] (0, 1.2) rectangle (\Largeur, 2.4);
        \node[font=\bfseries] at ({\Largeur/2}, 1.8) {#3};

        % Bloc des parts (Bas) : seule la première part est colorée
        \draw[thick, fill=#2!40] (0, 0) rectangle +(\w, 1.2);
        \ifnum#1>1
            \foreach \i in {2,...,#1} {
                \pgfmathsetmacro{\xpos}{(\i-1)*\w}
                \draw[thick, fill=white] (\xpos, 0) rectangle +(\w, 1.2);
            }
        \fi
        \node[font=\bfseries] at ({\w/2}, 0.6) {#4};
    \end{tikzpicture}%
}

% ==========================================
% MACROS : GABARITS VIDES (COLONNE DROITE)
% ==========================================
\newcommand{\gabaritVide}[2]{%
    \begin{tikzpicture}[scale=1]
        \useasboundingbox (0, -0.3) rectangle (\Largeur, 2.7);
        \pgfmathsetmacro{\w}{\Largeur/#1}

        % Bloc total (Haut) : ligne d'écriture basse, fine et claire
        \draw[thick, fill=#2!20] (0, 1.2) rectangle (\Largeur, 2.4);
        \draw[ligneecriture] ({\Largeur/2 - 1.5}, 1.58) -- ({\Largeur/2 + 1.5}, 1.58);

        % Bloc des parts (Bas) : seule la première part est colorée
        \draw[thick, fill=#2!40] (0, 0) rectangle +(\w, 1.2);
        \ifnum#1>1
            \foreach \i in {2,...,#1} {
                \pgfmathsetmacro{\xpos}{(\i-1)*\w}
                \draw[thick, fill=white] (\xpos, 0) rectangle +(\w, 1.2);
            }
        \fi
        \pgfmathsetmacro{\dotw}{min(\w*0.30, 0.65)}
        \draw[ligneecriture] ({\w/2-\dotw}, 0.38) -- ({\w/2+\dotw}, 0.38);
    \end{tikzpicture}%
}



% ==========================================
% MACROS : GABARITS MULTIPLES (PAGE 2)
% Une seule part est donnée en trait plein ;
% les autres parts du multiple sont suggérées en pointillés,
% sans recouvrir les traits pleins déjà porteurs de sens.
% ==========================================
\newcommand{\gabaritMultipleRempli}[4]{%
    \begin{tikzpicture}[scale=1]
        \useasboundingbox (0, -0.3) rectangle (\Largeur, 2.7);
        \pgfmathsetmacro{\w}{\Largeur/#1}

        % Bloc du multiple (Haut) : contour plein sur toute la largeur
        \draw[thick, fill=#2!20] (0, 1.2) rectangle (\Largeur, 2.4);
        \node[font=\bfseries] at ({\Largeur/2}, 1.8) {#3};

        % Une part connue en trait plein
        \draw[thick, fill=#2!40] (0, 0) rectangle +(\w, 1.2);
        \node[font=\bfseries] at ({\w/2}, 0.6) {#4};

        % Parts à imaginer : pointillés uniquement sur les bords non déjà tracés en plein
        % La séparation avec la part pleine et la limite avec le multiple restent pleines.
        \ifnum#1>1
            \draw[pointillesforts] (\w, 0) -- (\Largeur, 0);
            \foreach \i in {2,...,#1} {
                \pgfmathsetmacro{\xpos}{\i*\w}
                \draw[pointillesforts] (\xpos, 0) -- (\xpos, 1.2);
            }
        \fi
    \end{tikzpicture}%
}

\newcommand{\gabaritMultipleVide}[2]{%
    \begin{tikzpicture}[scale=1]
        \useasboundingbox (0, -0.3) rectangle (\Largeur, 2.7);
        \pgfmathsetmacro{\w}{\Largeur/#1}

        % Bloc du multiple (Haut) : contour plein, réponse à compléter
        \draw[thick, fill=#2!20] (0, 1.2) rectangle (\Largeur, 2.4);
        \draw[ligneecriture] ({\Largeur/2 - 1.5}, 1.58) -- ({\Largeur/2 + 1.5}, 1.58);

        % Une part en trait plein, à compléter par une ligne d'écriture basse et claire
        \draw[thick, fill=#2!40] (0, 0) rectangle +(\w, 1.2);
        \pgfmathsetmacro{\dotw}{min(\w*0.25, 0.5)}
        \draw[ligneecriture] ({\w/2-\dotw}, 0.38) -- ({\w/2+\dotw}, 0.38);

        % Parts à imaginer : pointillés uniquement sur les bords non déjà tracés en plein
        % La séparation avec la part pleine et la limite avec le multiple restent pleines.
        \ifnum#1>1
            \draw[pointillesforts] (\w, 0) -- (\Largeur, 0);
            \foreach \i in {2,...,#1} {
                \pgfmathsetmacro{\xpos}{\i*\w}
                \draw[pointillesforts] (\xpos, 0) -- (\xpos, 1.2);
            }
        \fi
    \end{tikzpicture}%
}
