% #1 ticks (-1 = none), #2 y-space before board
\newcommand{\drawBoard}[2]{%
\vspace*{#2}
\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm]
  \def\xstart{5.55}
  \def\L{18.7}
  \def\ytop{1.80}
  \def\ybot{-1.80}
  \ifnum#1>0
    \pgfmathsetmacro{\dx}{\L/(#1)}
  \fi

  % Larger soft colored bands, without border, to leave more writing space
  \fill[montorange,rounded corners=5pt] (5.05,0.72) rectangle (25.00,2.88);
  \fill[montblue,rounded corners=5pt] (5.05,-2.88) rectangle (25.00,-0.72);

  % Label bubbles on the left, aligned with each graduated line.
  \fill[montorange,rounded corners=6pt] (0.05,\ytop-1.65) rectangle (4.35,\ytop+1.45);
  \draw[toporange,line width=0.8pt,rounded corners=6pt] (0.05,\ytop-1.65) rectangle (4.35,\ytop+1.45);
  \node[anchor=west,font=\large\bfseries,text=toporange] at (0.30,\ytop+1.02) {Grandeur 1 :};
  \draw[softgray,densely dotted,line width=0.55pt] (0.30,\ytop+0.22)--(3.20,\ytop+0.22);
  \node[anchor=west,font=\large\bfseries,text=toporange] at (0.30,\ytop-0.58) {Unité de mesure :};
  \draw[softgray,densely dotted,line width=0.55pt] (0.30,\ytop-1.38)--(3.20,\ytop-1.38);

  \fill[montblue,rounded corners=6pt] (0.05,\ybot-1.65) rectangle (4.35,\ybot+1.45);
  \draw[bottomblue,line width=0.8pt,rounded corners=6pt] (0.05,\ybot-1.65) rectangle (4.35,\ybot+1.45);
  \node[anchor=west,font=\large\bfseries,text=bottomblue] at (0.30,\ybot+1.02) {Grandeur 2 :};
  \draw[softgray,densely dotted,line width=0.55pt] (0.30,\ybot+0.22)--(3.20,\ybot+0.22);
  \node[anchor=west,font=\large\bfseries,text=bottomblue] at (0.30,\ybot-0.58) {Unité de mesure :};
  \draw[softgray,densely dotted,line width=0.55pt] (0.30,\ybot-1.38)--(3.20,\ybot-1.38);

  \draw[toporange,line width=1.5pt,-stealth] (\xstart,\ytop)--(\xstart+\L+0.35,\ytop);
  \draw[bottomblue,line width=1.5pt,-stealth] (\xstart,\ybot)--(\xstart+\L+0.35,\ybot);

  \ifnum#1>0
    \foreach \i in {0,...,#1}{
      \pgfmathsetmacro{\x}{\xstart + \i*\dx}
      \draw[toporange,line width=0.8pt] (\x,\ytop-0.14)--(\x,\ytop+0.14);
      \draw[bottomblue,line width=0.8pt] (\x,\ybot-0.14)--(\x,\ybot+0.14);
    }
  \fi
\end{tikzpicture}
\end{center}
}
